/** * ==================================================================== * BLANGKO KUNJUNGAN POLMAS "PRESISI WARGA NTB" — Backend (Apps Script) * ==================================================================== * CARA SETUP: * 1. Buat Google Spreadsheet baru, beri nama misalnya "Data Kunjungan Polmas NTB". * 2. Salin ID Spreadsheet-nya (bagian di URL antara /d/ dan /edit), * lalu tempel di CONFIG.SHEET_ID di bawah ini. * 3. Buka Extensions > Apps Script dari Spreadsheet tersebut. * 4. Hapus isi Code.gs bawaan, tempel SELURUH isi file ini. * 5. Klik Deploy > New deployment > pilih tipe "Web app". * - Execute as: Me * - Who has access: Anyone * 6. Salin URL yang diberikan (diakhiri /exec), tempel ke CONFIG.API_URL * di file index.html. * 7. Setiap kali mengubah kode ini, deploy ulang dengan memilih * "New deployment" atau "Manage deployments" > Edit > Version: New. * ==================================================================== */ const CONFIG = { SHEET_ID: '1mcZjrZ7E1JjEEGc4IsDzE9m2BboWV6HIq78IlU-FPf8', SHEET_NAME: 'DataKunjungan', FOLDER_NAME: 'Dokumentasi Polmas NTB' }; const HEADERS = [ 'Timestamp', 'ID Kunjungan', 'Tanggal Kunjungan', 'Nama Petugas', 'Warga Ke', 'Nama Warga', 'Jenis Kelamin', 'Umur', 'Pekerjaan', 'No Telp', 'Alamat', 'Aspek', 'Informasi/Saran/Permasalahan/Keluhan', 'Tindak Lanjut', 'Nama Ketua RT/RW/Ka-Dus/Ka-Ling', 'Foto Dokumentasi', 'TTD Petugas', 'TTD Warga', 'TTD Ketua RT/RW' ]; // -------------------------------------------------------------------- // ENTRY POINTS // -------------------------------------------------------------------- function doPost(e) { let result; try { const data = JSON.parse(e.postData.contents); if (data.action === 'submit') { result = handleSubmit(data); } else if (data.action === 'list') { result = { ok: true, rows: handleList(data.limit || 200) }; } else if (data.action === 'get') { result = { ok: true, rows: handleGet(data.idKunjungan) }; } else { result = { ok: false, error: 'Aksi tidak dikenali.' }; } } catch (err) { result = { ok: false, error: String(err) }; } return ContentService.createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } function doGet(e) { const action = e.parameter.action; let result; try { if (action === 'list') { result = { ok: true, rows: handleList(200) }; } else if (action === 'get') { result = { ok: true, rows: handleGet(e.parameter.idKunjungan) }; } else { result = { ok: true, status: 'Polmas API aktif.' }; } } catch (err) { result = { ok: false, error: String(err) }; } return ContentService.createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } // -------------------------------------------------------------------- // HANDLERS // -------------------------------------------------------------------- function handleSubmit(data) { const sheet = getSheet(); const now = new Date(); const idKunjungan = 'PLM-' + Utilities.formatDate(now, 'GMT+8', 'yyyyMMdd-HHmmss'); // TTD Petugas hanya disimpan sekali, dipakai untuk semua baris warga. const ttdPetugasUrl = data.ttdPetugas ? saveImage(data.ttdPetugas, idKunjungan + '-ttd-petugas') : ''; const wargaList = data.warga || []; const savedRows = []; wargaList.forEach(function (w, idx) { const fotoUrls = (w.fotos || []).map(function (foto, i) { return saveImage(foto, idKunjungan + '-warga' + w.label + '-foto' + (i + 1)); }).join(', '); const ttdWargaUrl = w.ttdWarga ? saveImage(w.ttdWarga, idKunjungan + '-warga' + w.label + '-ttd-warga') : ''; const ttdKetuaUrl = w.ttdKetuaRT ? saveImage(w.ttdKetuaRT, idKunjungan + '-warga' + w.label + '-ttd-ketua') : ''; const row = [ now, idKunjungan, data.tanggal || '', data.namaPetugas || '', w.label || (idx === 0 ? 'I' : 'II'), w.nama || '', w.gender || '', w.umur || '', w.pekerjaan || '', w.telp || '', w.alamat || '', (w.aspek || []).join(', '), w.informasi || '', w.tindakLanjut || '', w.namaKetuaRT || '', fotoUrls, ttdPetugasUrl, ttdWargaUrl, ttdKetuaUrl ]; sheet.appendRow(row); savedRows.push(row); }); return { ok: true, idKunjungan: idKunjungan }; } function handleList(limit) { const sheet = getSheet(); const values = sheet.getDataRange().getValues(); if (values.length < 2) return []; const rows = values.slice(1).map(rowToObject); rows.reverse(); return rows.slice(0, limit); } /** * Dipakai saat "Cetak Blangko" dari Riwayat: mengambil baris untuk satu * kunjungan, dan mengubah SEMUA foto/ttd (yang di Sheet cuma berupa ID file * Drive) menjadi data base64 langsung di dalam JSON. Dengan begitu gambar * tidak lagi bergantung pada link Drive terpisah yang kadang gagal dimuat * browser tanpa sesi login — datanya sudah menyatu di respons ini. */ function handleGet(idKunjungan) { const sheet = getSheet(); const values = sheet.getDataRange().getValues(); if (values.length < 2) return []; const rows = values.slice(1) .map(rowToObject) .filter(function (r) { return r.idKunjungan === idKunjungan; }); rows.forEach(function (r) { r.ttdPetugas = fileIdToDataUri(r.ttdPetugas); r.ttdWarga = fileIdToDataUri(r.ttdWarga); r.ttdKetuaRT = fileIdToDataUri(r.ttdKetuaRT); r.foto = (r.foto || '') .split(',') .map(function (s) { return s.trim(); }) .filter(Boolean) .map(fileIdToDataUri) .join(','); }); return rows; } /** Ubah 1 ID file Drive menjadi data URI base64 (data:image/...;base64,...). */ function fileIdToDataUri(fileId) { if (!fileId) return ''; try { const blob = DriveApp.getFileById(fileId).getBlob(); return 'data:' + blob.getContentType() + ';base64,' + Utilities.base64Encode(blob.getBytes()); } catch (err) { return ''; } } function rowToObject(row) { return { timestamp: row[0], idKunjungan: row[1], tanggal: row[2], namaPetugas: row[3], wargaKe: row[4], nama: row[5], gender: row[6], umur: row[7], pekerjaan: row[8], telp: row[9], alamat: row[10], aspek: row[11], informasi: row[12], tindakLanjut: row[13], namaKetuaRT: row[14], foto: row[15], ttdPetugas: row[16], ttdWarga: row[17], ttdKetuaRT: row[18] }; } // -------------------------------------------------------------------- // UTIL: SHEET & DRIVE // -------------------------------------------------------------------- function getSheet() { const ss = SpreadsheetApp.openById(CONFIG.SHEET_ID); let sheet = ss.getSheetByName(CONFIG.SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(CONFIG.SHEET_NAME); } if (sheet.getLastRow() === 0) { sheet.appendRow(HEADERS); sheet.setFrozenRows(1); } return sheet; } function getFolder() { const folders = DriveApp.getFoldersByName(CONFIG.FOLDER_NAME); if (folders.hasNext()) return folders.next(); return DriveApp.createFolder(CONFIG.FOLDER_NAME); } /** * Menyimpan gambar base64 (data URL) ke Drive dan mengembalikan ID file-nya * saja (bukan link). ID inilah yang disimpan di Sheet; saat data ditampilkan * lagi (Riwayat/Cetak Ulang), ID ini diubah kembali jadi gambar base64 * langsung lewat fileIdToDataUri() — jadi tidak pernah bergantung pada link * Drive yang bisa gagal dimuat browser. */ function saveImage(dataUrl, filename) { const match = dataUrl.match(/^data:(image\/[a-zA-Z0-9.+-]+);base64,(.*)$/); if (!match) return ''; const contentType = match[1]; const base64 = match[2]; const ext = contentType.split('/')[1] === 'jpeg' ? 'jpg' : contentType.split('/')[1]; const bytes = Utilities.base64Decode(base64); const blob = Utilities.newBlob(bytes, contentType, filename + '.' + ext); const file = getFolder().createFile(blob); return file.getId(); }
